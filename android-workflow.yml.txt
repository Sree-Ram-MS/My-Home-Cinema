# Copy this file to .github/workflows/android.yml in the GitHub web UI
# (the Muse GitHub App cannot create workflow files directly).
# v3: v2 build (no setup-android step, runner SDK + sdkmanager) PLUS
# automatic Releases. Every push to android still uploads the APK as
# an Actions artifact. When you push a tag like v1.0, the same build
# also creates a GitHub Release for that tag and attaches the APK.
name: Build Android APK

permissions:
  contents: write

on:
  push:
    branches: [android]
    tags: ['v*']
  workflow_dispatch:

jobs:
  build:
    runs-on: ubuntu-latest
    steps:
      - name: Checkout
        uses: actions/checkout@v4
      - name: Set up JDK 17
        uses: actions/setup-java@v5
        with:
          distribution: temurin
          java-version: '17'
      - name: Show preinstalled Android SDK
        run: |
          echo "ANDROID_HOME=$ANDROID_HOME"
          ls "$ANDROID_HOME" || true
          ls "$ANDROID_HOME/cmdline-tools" || true
      - name: Install SDK packages
        run: |
          set -e
          SDKMANAGER=""
          if [ -x "$ANDROID_HOME/cmdline-tools/latest/bin/sdkmanager" ]; then
            SDKMANAGER="$ANDROID_HOME/cmdline-tools/latest/bin/sdkmanager"
          elif [ -x "$ANDROID_HOME/cmdline-tools/bin/sdkmanager" ]; then
            SDKMANAGER="$ANDROID_HOME/cmdline-tools/bin/sdkmanager"
          else
            echo "Downloading Android cmdline-tools..."
            mkdir -p "$ANDROID_HOME/cmdline-tools"
            cd /tmp
            curl -L -o cmdtools.zip https://dl.google.com/android/repository/commandlinetools-linux-11076708_latest.zip
            unzip -q -o cmdtools.zip -d "$ANDROID_HOME/cmdline-tools/tmp"
            mkdir -p "$ANDROID_HOME/cmdline-tools/latest"
            mv "$ANDROID_HOME/cmdline-tools/tmp/cmdline-tools/"* "$ANDROID_HOME/cmdline-tools/latest/"
            SDKMANAGER="$ANDROID_HOME/cmdline-tools/latest/bin/sdkmanager"
          fi
          echo "sdkmanager: $SDKMANAGER"
          yes | "$SDKMANAGER" --licenses || true
          yes | "$SDKMANAGER" "platforms;android-34" "build-tools;34.0.0" "platform-tools"
      - name: Build debug APK with Gradle
        uses: gradle/gradle-build-action@v3
        with:
          gradle-version: '8.5'
          arguments: assembleDebug --stacktrace
      - name: Upload APK artifact
        uses: actions/upload-artifact@v4
        with:
          name: My-Home-Cinema-debug-apk
          path: app/build/outputs/apk/debug/app-debug.apk
      - name: Prepare release APK name
        if: startsWith(github.ref, 'refs/tags/v')
        run: cp app/build/outputs/apk/debug/app-debug.apk "My-Home-Cinema-${{ github.ref_name }}.apk"
      - name: Create Release and attach APK
        if: startsWith(github.ref, 'refs/tags/v')
        uses: softprops/action-gh-release@v2
        with:
          tag_name: ${{ github.ref_name }}
          name: My Home Cinema Android ${{ github.ref_name }}
          body: |
            My Home Cinema Android app — start sharing on your phone, open the shown IP:port on your TV browser.
            Install the APK below on your Android phone, allow video/notification permissions, tap Start Sharing.
          files: My-Home-Cinema-${{ github.ref_name }}.apk
          generate_release_notes: true
